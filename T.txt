state.manualIsCooldown = false;
state.manualCooldownEnd = 0;
showFeedback('🐕 Watchdog: stale manual cooldown cleared');

}

if (state.globalCooldownActive && state.globalCooldownEnd > 0 && now > state.globalCooldownEnd) {
state.globalCooldownActive = false;
state.globalCooldownEnd = 0;
showFeedback('🐕 Watchdog: stale global cooldown cleared');
recovered = true;
}

if (state.isCurrentlySelecting) {
const stuckFor = state.isCurrentlySelectingSetAt > 0 ? (now - state.isCurrentlySelectingSetAt) : 0;
if (stuckFor > 30000) {
state.isCurrentlySelecting = false;
state.isCurrentlySelectingSetAt = 0;
showFeedback('🐕 Watchdog: unstuck isCurrentlySelecting');
recovered = true;
}
}

if (!isInCooldown()) {
const FREEZE_THRESHOLD = 30000;

if (state.minuteMarkActive && state.minuteMarkEndTime > now) {
const timeSinceFire = now - state.lastMinuteMarkSelection;
if (timeSinceFire > FREEZE_THRESHOLD) {
console.warn('[Watchdog] MinuteMark window active but no fire in 30s — recovering');
softRecoverSelection('minuteMark');
recovered = true;
}
}

if (state.scheduledIsActive && state.scheduledEndTime > now) {
const timeSinceFire = now - state.lastScheduledSelection;
if (timeSinceFire > FREEZE_THRESHOLD) {
console.warn('[Watchdog] Scheduled window active but no fire in 30s — recovering');
softRecoverSelection('scheduled');
recovered = true;
}
}
}

if (recovered) {
state.watchdogLastFired = now;
saveState();
updateUI();
}
};

const calculateNextScheduledRun = () => {

const now = new Date();
let nextRun = null;
for (const hour of CONFIG.scheduledHours) {

const [minutesStr, secondsStr = "0"] = timeStr.split(':');
const scheduledMinute = parseInt(minutesStr);
const scheduledSecond = parseInt(secondsStr);
const scheduledTime = new Date();
scheduledTime.setHours(hour, scheduledMinute, scheduledSecond, 0);
if (scheduledTime <= now) scheduledTime.setDate(scheduledTime.getDate() + 1);
if (!nextRun || scheduledTime < nextRun) nextRun = scheduledTime;
}
}
state.nextScheduledRun = nextRun ? nextRun.getTime() : 0;
saveState();
};

const scheduledMinute = parseInt(minutesStr);
const scheduledSecond = parseInt(secondsStr);

// === CUSTOM WINDOW RELOAD — high-precision rAF loop ===

const _customWindowTick = () => {
if (!CONFIG.customWindowEnabled) {
_cwRafId = requestAnimationFrame(_customWindowTick);
return;
}

const now = new Date();
const currentMinute = now.getMinutes();
const currentSecond = now.getSeconds();

const targetMinutes = String(CONFIG.customWindowMinutes)
.split(',').map(s => parseInt(s.trim())).filter(n => !isNaN(n));

if (!targetMinutes.includes(currentMinute)) {
_cwFired.clear();
_cwRafId = requestAnimationFrame(_customWindowTick);
return;
}

if (!isInCooldown()) {
const w1s  = CONFIG.customWindowW1Start;
const w1e  = CONFIG.customWindowW1End;
const iv1  = CONFIG.customWindowW1Interval || 4;
const w2s  = CONFIG.customWindowW2Start;
const w2e  = CONFIG.customWindowW2End;
const iv2  = CONFIG.customWindowW2Interval || 3;

const inW1 = currentSecond >= w1s && currentSecond <= w1e && (currentSecond - w1s) % iv1 === 0;
const inW2 = currentSecond >= w2s && currentSecond <= w2e && (currentSecond - w2s) % iv2 === 0;

if (inW1 || inW2) {
const fireKey = currentMinute * 100 + currentSecond;
if (!_cwFired.has(fireKey)) {
_cwFired.add(fireKey);
minuteMarkSelectPost();
showFeedback(`⚡ W${inW1 ? 1 : 2} M${currentMinute}:${String(currentSecond).padStart(2,'0')} → reload`);
}
}
}

_cwRafId = requestAnimationFrame(_customWindowTick);
};

_cwFired.clear();
_cwRafId = requestAnimationFrame(_customWindowTick);
};
const stopCustomWindowLoop = () => {
if (_cwRafId) { cancelAnimationFrame(_cwRafId); _cwRafId = null; }
_cwFired.clear();
};

const checkRandomSlots = () => {
if (!CONFIG.randomSlotSelectionEnabled || state.manualIsRunning) return;
if (isInCooldown()) return;

const now = new Date();
const currentMinute = now.getMinutes();
const currentSecond = now.getSeconds();
const totalSeconds = (currentMinute * 60) + currentSecond;

const intervalKey = Math.floor(totalSeconds / RANDOM_INTERVAL);

if (SKIPPED_MINUTES.includes(currentMinute)) {
if (intervalKey !== state.lastRandomIntervalKey) {
state.lastRandomIntervalKey = intervalKey;
state.randomTriggered = false;
saveState();
}
return;
}

if (intervalKey !== state.lastRandomIntervalKey) {
if (CONFIG.randomSlot === "manual") {
const [startStr, endStr] = CONFIG.manualRandomSlot.split('-');
const start = parseInt(startStr);
const end = parseInt(endStr);
if (!isNaN(start) && !isNaN(end) && start >= 0 && end <= RANDOM_INTERVAL && start < end) {
state.randomSecond = Math.floor(Math.random() * (end - start + 1)) + start;
} else {
state.randomSecond = 135 + Math.floor(Math.random() * 6);
}
} else {
const [startStr, endStr] = CONFIG.randomSlot.split('-');
const start = parseInt(startStr);
const end = parseInt(endStr);
state.randomSecond = Math.floor(Math.random() * (end - start + 1)) + start;
}
state.lastRandomIntervalKey = intervalKey;
state.randomTriggered = false;
saveState();
}

if (!state.randomTriggered) {
const positionInBlock = totalSeconds % RANDOM_INTERVAL;
if (positionInBlock >= state.randomSecond) {
if (!isInCooldown()) {
randomSelectPost();
state.randomTriggered = true;
saveState();
}
}
}
};

const shareSettingsViaAbly = () => {
try {
  const ablyPub = new Ably.Realtime(ABLY_KEY);
  ablyPub.connection.once('connected', () => {
    const ch = ablyPub.channels.get(ABLY_CHANNEL);
    const payload = {
      type:      'DEVICE_SETTINGS',
      deviceId:  _deviceId,
      timestamp: Date.now(),
      config:    JSON.parse(JSON.stringify(CONFIG))
    };
    ch.publish('device_settings', payload, (err) => {
      if (err) {
        showFeedback('❌ Share failed: ' + err.message);
        console.error('[SHARE] publish error', err);
      } else {
        showFeedback('📡 Settings shared! (' + _deviceId + ')');
        flashScreen('#f9a825', 600);
      }
      ablyPub.close();
    });
  });
  ablyPub.connection.once('failed', () => {
    showFeedback('❌ Ably connect failed');
  });
} catch(e) {
  showFeedback('❌ Share error: ' + e.message);
  console.error('[SHARE]', e);
}
};

// ABLY REMOTE CONTROL — SIGNAL RECEIVER

const setupAblyReceiver = () => {
try {
const ably = new Ably.Realtime(ABLY_KEY);
const channel = ably.channels.get(ABLY_CHANNEL);

// Listen for push-config from control panel
channel.subscribe('push_config', (message) => {
  const data = message.data;
  if (data.target !== 'ALL' && data.target !== _deviceId) return;
  console.log('[RECEIVER] push_config received for', data.target);
  CONFIG = {...DEFAULT_CONFIG, ...CONFIG, ...data.config};
  saveConfig();
  applyPanelTheme();
  restartScript();
  showFeedback('🔄 Config synced from control panel!');
  showRemoteToast('🔄 Config synced!', 'linear-gradient(135deg,#00897b 0%,#004d40 100%)');
  flashScreen('#00897b', 800);
});

channel.subscribe('msg', (message) => {
const signal = message.data;
console.log('[RECEIVER] Got signal:', signal);

if (signal === 'RESELECT_500') {
CONFIG.minuteMarkReselectInterval = 500;
saveConfig();
updateIntervals();
showFeedback('⚡ Remote: 500ms interval applied!');
showRemoteToast('⚡ 500ms Interval Applied!', 'linear-gradient(135deg, #1565c0 0%, #0d47a1 100%)');
flashScreen('#1565c0', 800);
updateUI();
console.log('[RECEIVER] minuteMarkReselectInterval → 500ms');

} else if (signal === 'RESELECT_2000') {
CONFIG.minuteMarkReselectInterval = 2000;
saveConfig();
updateIntervals();
showFeedback('🔵 Remote: 2000ms interval applied!');
showRemoteToast('🔵 2000ms Interval Applied!', 'linear-gradient(135deg, #2e7d32 0%, #1b5e20 100%)');
flashScreen('#2e7d32', 800);
updateUI();
console.log('[RECEIVER] minuteMarkReselectInterval → 2000ms');

} else if (signal === 'RESELECT_3000') {
CONFIG.minuteMarkReselectInterval = 3000;
saveConfig();
updateIntervals();
showFeedback('🟡 Remote: 3000ms interval applied!');
showRemoteToast('🟡 3000ms Interval Applied!', 'linear-gradient(135deg, #e65100 0%, #bf360c 100%)');
flashScreen('#e65100', 800);
updateUI();
console.log('[RECEIVER] minuteMarkReselectInterval → 3000ms');

} else if (signal === 'KARACHI_ON') {
switchToKarachi();
showRemoteToast('📍 Karachi Post ON', 'linear-gradient(135deg, #880e4f 0%, #ad1457 100%)');

} else if (signal === 'KARACHI_OFF') {
switchToIslamabad();
showRemoteToast('📍 Karachi Post OFF', '#555');

} else if (signal === 'ISLAMABAD_ON') {
switchToIslamabad();
showRemoteToast('📍 Islamabad Post ON', 'linear-gradient(135deg, #4a148c 0%, #6a1b9a 100%)');

} else if (signal === 'ISLAMABAD_OFF') {
showRemoteToast('📍 Islamabad Post OFF', '#555');
}
});

ably.connection.on('connected', () => {
console.log('[RECEIVER] Ably connected — listening for signals');
});

ably.connection.on('failed', () => {
console.warn('[RECEIVER] Ably connection failed');
});

} catch (e) {
console.error('[RECEIVER] Ably setup error:', e);
}
};

saveState();
updateUI();

}, state.globalCooldownEnd - now);
}

if (state.manualCooldownEnd > now) {

state.manualCooldownEnd = 0;
saveState();
updateUI();
if (!isInCooldown()) resetPostSelection();
}, state.manualCooldownEnd - now);
}

state.scheduledIsActive = true;
const remaining = state.scheduledEndTime - now;
scheduledSelectPost();
state.scheduledIntervalId = setInterval(() => {
if (!state.scheduledIsActive) return;
if (isInCooldown()) { stopScheduledSelection(); return; }
if (Date.now() - state.lastScheduledSelection >= CONFIG.scheduledReselectInterval) {
if (checkForAvailableDates()) { stopScheduledSelection(); return; }
scheduledSelectPost();
}
}, 1000);
setTimeout(() => { stopScheduledSelection(); showFeedback('Scheduled selection period ended'); }, remaining);
}

if (state.minuteMarkEndTime > now) {
state.minuteMarkActive = true;
const remaining = state.minuteMarkEndTime - now;
minuteMarkSelectPost();
state.minuteMarkIntervalId = setInterval(() => {
if (!state.minuteMarkActive) return;
if (isInCooldown()) { stopMinuteMarkSelection(); return; }
if (Date.now() - state.lastMinuteMarkSelection >= CONFIG.minuteMarkReselectInterval) {
if (checkForAvailableDates()) { stopMinuteMarkSelection(); return; }
minuteMarkSelectPost();
}

setTimeout(() => { stopMinuteMarkSelection(); showFeedback('Minute mark selection period ended'); }, remaining);

};
